96 private string ToTitleCase(string str)

99 bool newWord = true;
100
101 for (int i = 0; i < chars.Length; i++)
102 {
103 if (newWord && char.IsLetter(chars[i]))
104 {
105 chars[i] = char.ToUpper(chars[i]);
106 newWord = false;
107 }
108 else if (char.IsWhiteSpace(chars[i]) || chars[i] == '_' || chars[i] == '-')
109 {
110 newWord = true;
111 }
112 }
