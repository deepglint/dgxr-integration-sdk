194 internal bool PairDeviceToCharacter(Character character, InputDevice pairDevice)
195 {
196 PlayerInput playerInput = null;
197 if (character.m_player is null)
198 {
199 // Initiate a new player for the character and pair the device to the new player.
200 playerInput = playerInputManager.JoinPlayer(pairWithDevice: pairDevice);
201 Player player = playerInput.gameObject.AddComponent<Player>();
202 character.m_player = player;
203 player.m_Character = character;
204 player.m_PlayerInput = playerInput;
205 Debug.LogFormat("player {0} joined with character {1}", playerInput.user.id, character.Name);
206 return true;
207 }
208
209 playerInput = character.m_player.m_PlayerInput;
210 if (playerInput is not null)
211 {
212 // Pair the device to the player which controls current character.
213 InputUser.PerformPairingWithDevice(pairDevice, playerInput.user);
214 return true;
215 }
216
217 return false;
218 }

300 private static void OnInputUserChange(InputUser user, InputUserChange change, InputDevice device)
301 {
302 switch (change)

304 case InputUserChange.DeviceLost:
305 for (var i = 0; i < s_AllActivePlayersCount; ++i)
306 {
307 var player = s_AllActivePlayers[i];
308 if (player.user == user)
309 {
310 OnDeviceLost(player, device);

312 }
313
314 break;
315 case InputUserChange.DeviceRegained:
316 for (var i = 0; i < s_AllActivePlayersCount; ++i)
317 {
318 var player = s_AllActivePlayers[i];
319 if (player.user == user)
320 {
321 OnDeviceRegained(player, device);
322 }
323 }
324
325 break;
326 case InputUserChange.DevicePaired:
327 for (var i = 0; i < s_AllActivePlayersCount; ++i)
328 {
329 var player = s_AllActivePlayers[i];
