130 private void LoadSensorConfiguration(out Sensor sensor)
131 {
132 sensor = new Sensor();
133 var sensorsConfig = _xmlDoc.GetElementsByTagName("Sensor");
134 var sensorConfig = sensorsConfig[0].ChildNodes;
135 foreach (XmlNode parameter in sensorConfig)
136 {
137 switch (parameter.Name)
138 {
139 case "Type":
140 switch (parameter.InnerText)
141 {
142 case "DG":
143 sensor.Type = SensorType.DG;
144 break;
145 case "ARTTrack":
146 sensor.Type = SensorType.ArtTrack;
147 break;
148 default:
149 MDebug.Log("Loading the sensor type value: " + parameter.Name + " is an unknown type");
150 break;
151 }
152 break;
153 case "HeadTracker":
154 sensor.HeadTracker = string.Equals(parameter.InnerText, "true", StringComparison.InvariantCultureIgnoreCase);
155 break;
156 case "BodyID":
157 sensor.BodyID = int.Parse(parameter.InnerText);
158 break;
159 case "Stereo":
160 sensor.Stereo = string.Equals(parameter.InnerText, "true", StringComparison.InvariantCultureIgnoreCase);
161 break;
162 case "Pos.X":
163 sensor.Position.x = float.Parse(parameter.InnerText);
164 break;
165 case "Pos.Y":
166 sensor.Position.y = float.Parse(parameter.InnerText);
167 break;
168 case "Pos.Z":
169 sensor.Position.z = float.Parse(parameter.InnerText);
170 break;
171 case "Rot.X":
172 sensor.Rotation.x = float.Parse(parameter.InnerText);
173 break;
174 case "Rot.Y":
175 sensor.Rotation.y = float.Parse(parameter.InnerText);
176 break;
177 case "Rot.Z":
178 sensor.Rotation.z = float.Parse(parameter.InnerText);
179 break;
180 case "#comment":
181 break;
182 default:
183 MDebug.Log("Loading the sensor calibration values: " + parameter.Name + " is an unknown parameter");
184 break;
185 }
186 }
