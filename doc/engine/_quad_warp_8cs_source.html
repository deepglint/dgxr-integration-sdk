17 Matrix4x4 CalcHomography(Vector2 topLeft, Vector2 topRight, Vector2 bottomRight, Vector2 bottomLeft)
18 {
19 var sx = (topLeft.x - topRight.x) + (bottomRight.x - bottomLeft.x);
20 var sy = (topLeft.y - topRight.y) + (bottomRight.y - bottomLeft.y);
21
22 var dx1 = topRight.x - bottomRight.x;
23 var dx2 = bottomLeft.x - bottomRight.x;
24 var dy1 = topRight.y - bottomRight.y;
25 var dy2 = bottomLeft.y - bottomRight.y;
26
27 var z = (dx1 * dy2) - (dy1 * dx2);
28 var g = ((sx * dy2) - (sy * dx2)) / z;
29 var h = ((sy * dx1) - (sx * dy1)) / z;
30
31 var system = new[]
32 {
33 topRight.x - topLeft.x + g * topRight.x,
34 bottomLeft.x - topLeft.x + h * bottomLeft.x,
35 topLeft.x,
36 topRight.y - topLeft.y + g * topRight.y,
37 bottomLeft.y - topLeft.y + h * bottomLeft.y,
38 topLeft.y,
39 g,
40 h,
41 };
42
43 var mtx = Matrix4x4.identity;
44 mtx.m00 = system[0];
45 mtx.m01 = system[1];
46 mtx.m02 = system[2];
47 mtx.m10 = system[3];
48 mtx.m11 = system[4];
49 mtx.m12 = system[5];
