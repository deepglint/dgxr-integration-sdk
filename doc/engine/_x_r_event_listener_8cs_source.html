62 private Quaternion GetQuaternion(BodyDataSource data)
63 {
64 Vector3 leftShoulder = new Vector3(data.Joints[JointType.LeftShoulder].X,
65 data.Joints[JointType.LeftShoulder].Z,
66 data.Joints[JointType.LeftShoulder].Y);
67 Vector3 rightShoulder = new Vector3(data.Joints[JointType.RightShoulder].X,
68 data.Joints[JointType.RightShoulder].Z,
69 data.Joints[JointType.RightShoulder].Y);
70 Vector3 hip = new Vector3(
71 (data.Joints[JointType.LeftHip].X + data.Joints[JointType.RightHip].X) * 0.5f,
72 (data.Joints[JointType.LeftHip].Z + data.Joints[JointType.RightHip].Z) * 0.5f,
73 (data.Joints[JointType.LeftHip].Y + data.Joints[JointType.RightHip].Y) * 0.5f
74 );
75 Vector3 forward = -Vector3.Cross(leftShoulder - hip, rightShoulder - hip).normalized;
76 return Quaternion.LookRotation(forward);

356 private void OnPositionUpdate(string pId, Vector2 v2)
357 {
358 Vector2 position = new Vector2(v2.x + 1f, v2.y + 1f);
359 var state = new DGXRControllerState();
360 if (Math.Abs(v2.x) <= positionThresholdMin)
361 {
362 state.x = (byte)(127.5f);
363 } else if (v2.x >= positionThresholdMin + 1f)
364 {
365 state.x = (byte)(255);
366 } else if (v2.x <= -1 - positionThresholdMin)
367 {
368 state.x = (byte)(0);
369 }
370 else if (v2.x > 0)
371 {
372 state.x = (byte)((v2.x + 1f - positionThresholdMin) * 127.5f);
373 }
374 else
375 {
376 state.x = (byte)((v2.x + 1f + positionThresholdMin) * 127.5f);
377 }
378
379 if (Math.Abs(v2.y) <= positionThresholdMin)
380 {
381 state.y = (byte)(127.5f);
382 } else if (v2.y >= positionThresholdMin + 1f)
383 {
384 state.y = (byte)(255);
385 } else if (v2.y <= -1 - positionThresholdMin)
386 {
387 state.y = (byte)(0);
388 }
389 else if (v2.y > 0)
390 {
391 state.y = (byte)((v2.y + 1f - positionThresholdMin) * 127.5f);
392 }
393 else
394 {
395 state.y = (byte)((v2.y +1f + positionThresholdMin) * 127.5f);
396 }
397
398 //Debug.Log(pId + " x: "+ (float)state.x + ", y: " + (float)state.y);
399 InputDevice device = DeviceManager.GetActiveDeviceBySerial(pId);
400 if (device != null)
401 {
402 InputSystem.QueueStateEvent(device, state);
403 }

458 private void judgeRaiseHand()
459 {
460 foreach (var p1 in XRDGBodySource.Instance.Data)
461 {
462 if (p1.Value.Joints[JointType.RightHand].Z > p1.Value.Joints[JointType.HeadTop].Z)
463 {
464 if (RaiseHandOnQueue.ContainsKey(p1.Key))
465 {
466 RaiseHandOnQueue[p1.Key] += 1;
467 }
468 else
469 {
470 RaiseHandOnQueue[p1.Key] = 1;
471 }
472
473 RaiseHandOffQueue[p1.Key] = 0;
474 foreach (KeyValuePair<string, int> item in RaiseHandOnQueue)
475 {
476 if (item.Value >= 3)
477 {
478 if (RaiseHandResult.Contains(item.Key))
479 {
480 continue;
481 }
482 else
483 {
484 RaiseRightHandEventOn(item.Key);
485 RaiseHandResult.Add(item.Key);
486 }
487 }
488 }
489 }
490 else
491 {
492 if (RaiseHandResult.Contains(p1.Key))
