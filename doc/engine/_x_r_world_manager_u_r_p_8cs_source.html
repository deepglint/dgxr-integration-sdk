96 case XRLoadCalibration.SensorType.DG:

103 _sensor.transform.localRotation = Quaternion.Euler(_configuration.Sensors.Rotation.x, _configuration.Sensors.Rotation.y, _configuration.Sensors.Rotation.z);

115 _surfaces[index].transform.localRotation = Quaternion.Euler(surface.Rotation.x, surface.Rotation.y, surface.Rotation.z);
116 _surfaces[index].transform.Rotate(new Vector3(0, 180, 0));
117 _surfaces[index].transform.localScale = new Vector3(surface.Size.x / 10, surface.Size.y / 10, surface.Size.z / 10);
118 _surfaces[index].layer = _caveLayer;
119 _surfaceTextures[index] = new RenderTexture(_textureWidth, _textureHeight, 24, RenderTextureFormat.ARGB32);
120 _surfaceTextures[index].antiAliasing = 2;
121 _surfaceTextures[index].Create();
122 index++;
123 }

125 //Instantiate the CAVE projectors:
126 index = 0;
127 _projectors = new Camera[_configuration.Projectors.Length];
128 foreach (var projector in _configuration.Projectors)

139 _projectors[index].farClipPlane = (_projectors[index].transform.position - _surfaces[_projectors[index].targetDisplay].transform.position).magnitude * SpaceScale * 2;

141 SetObliqueness(0, projector.Fy, _projectors[index]);
142 _projectors[index].gameObject.layer = _caveLayer;
143 //_projectors[index].cullingMask = 1 << (_caveLayer - projector.Display);

151 _projectors[index].gameObject.GetComponent<QuadWarp>()._tex.Add(_surfaceTextures[surfaceIndex]);

223 public Vector3 GetHeadPosition()
224 {
225 if (_head == null) return Vector3.zero;
226 Vector3 scaleHead = _head.transform.localPosition * DisplayData.SpatialProportion;
227
228 // 基于空间点的移动偏移
229 float y = DisplayData.HumanEye + (DisplayData.HumanEye - scaleHead.z) * DisplayData.SpaceFollowSpeed;

264 private void SetHeadFovAndOrientationProjector(int index, Camera[] cameras, GameObject[,] edges)
265 {
266 var bottomToTop = edges[index, 0].transform.position - edges[index, 2].transform.position;
267 var leftToRight = edges[index, 1].transform.position - edges[index, 3].transform.position;
268
269 //Set FOV
270 cameras[index].ResetProjectionMatrix();
271 cameras[index].fieldOfView = 2 * Mathf.Rad2Deg *
272 Mathf.Atan(bottomToTop.magnitude / 2 /
273 (cameras[index].transform.localPosition.y *
274 cameras[index].transform.parent.lossyScale.y));
275
276 //Set the orientation
277 float obV = cameras[index].transform.localPosition.z *
278 cameras[index].transform.parent.lossyScale.z / (bottomToTop.magnitude / 2);
279 float obH = cameras[index].transform.localPosition.x *
280 cameras[index].transform.parent.lossyScale.x / (leftToRight.magnitude / 2);
281 SetObliqueness(obH, obV, cameras[index]);
282 }
283
284 private void SetHeadFovAndOrientationScreen(int index, Camera[] cameras, GameObject[,] edges)
285 {
286 var bottomToTop = edges[index, 0].transform.position - edges[index, 2].transform.position;
287 var leftToRight = edges[index, 1].transform.position - edges[index, 3].transform.position;
288
289 //Set FOV
290 cameras[index].ResetProjectionMatrix();
291 cameras[index].fieldOfView = -2 * Mathf.Rad2Deg *
292 Mathf.Atan(bottomToTop.magnitude / 2 /
293 (cameras[index].transform.localPosition.z *
294 cameras[index].transform.parent.lossyScale.z));
295
296 //Set the orientation
297 float obV = cameras[index].transform.localPosition.y *
298 cameras[index].transform.parent.lossyScale.y / (bottomToTop.magnitude / 2);
299 float obH = cameras[index].transform.localPosition.x *
300 cameras[index].transform.parent.lossyScale.x / (leftToRight.magnitude / 2);
301 SetObliqueness(-obH, -obV, cameras[index]);
302 }
