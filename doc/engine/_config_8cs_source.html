139 public bool SentConfig(List<Action> conf)
140 {
141 GameObject source = GameObject.Find("Source");
142 Source sourceConnect = source.GetComponent<Source>();
143 MDebug.LogFlow("2. 动作配置 - 2.0 权限 " + sourceConnect.HasConnectSuccess);
144 if (!sourceConnect.HasConnectSuccess)
145 {
146 return false;
147 }
148
149 if (sourceConnect != null && sourceConnect.webSocket != null)
150 {
151 string jsonString = JsonConvert.SerializeObject(conf);
152 sourceConnect.webSocket.Send(jsonString);
153 MDebug.LogFlow("2. 动作配置 - 2.1 发送成功");
154 MDebug.LogFlow("2. 动作配置 - 2.2 动作数量：" + conf.Count);
155 return true;
156 }
157
158 return false;
