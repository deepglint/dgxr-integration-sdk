272 private void SetHeadFovAndOrientationProjector(int index, Camera[] cameras, GameObject[,] edges)
273 {
274 var bottomToTop = edges[index, 0].transform.position - edges[index, 2].transform.position;
275 var leftToRight = edges[index, 1].transform.position - edges[index, 3].transform.position;
276
277 //Set FOV
278 cameras[index].ResetProjectionMatrix();
279 cameras[index].fieldOfView = 2 * Mathf.Rad2Deg *
280 Mathf.Atan(bottomToTop.magnitude / 2 /
281 (cameras[index].transform.localPosition.y *
282 cameras[index].transform.parent.lossyScale.y));
283
284 //Set the orientation
285 float obV = cameras[index].transform.localPosition.z *
286 cameras[index].transform.parent.lossyScale.z / (bottomToTop.magnitude / 2);
287 float obH = cameras[index].transform.localPosition.x *
288 cameras[index].transform.parent.lossyScale.x / (leftToRight.magnitude / 2);
289 SetObliqueness(obH, obV, cameras[index]);
290 }
291
292 private void SetHeadFovAndOrientationScreen(int index, Camera[] cameras, GameObject[,] edges)
293 {
294 var bottomToTop = edges[index, 0].transform.position - edges[index, 2].transform.position;
295 var leftToRight = edges[index, 1].transform.position - edges[index, 3].transform.position;
296
297 //Set FOV
298 cameras[index].ResetProjectionMatrix();
299 cameras[index].fieldOfView = -2 * Mathf.Rad2Deg *
300 Mathf.Atan(bottomToTop.magnitude / 2 /
301 (cameras[index].transform.localPosition.z *
302 cameras[index].transform.parent.lossyScale.z));
303
304 //Set the orientation
305 float obV = cameras[index].transform.localPosition.y *
306 cameras[index].transform.parent.lossyScale.y / (bottomToTop.magnitude / 2);
307 float obH = cameras[index].transform.localPosition.x *
308 cameras[index].transform.parent.lossyScale.x / (leftToRight.magnitude / 2);
309 SetObliqueness(-obH, -obV, cameras[index]);
310 }
