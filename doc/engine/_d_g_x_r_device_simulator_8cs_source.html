81 if (keyboard == null)

83
84 var state = new DGXRControllerState();
85
86 state.x = 127;
87 state.y = 127;
88
89 // Map WASD to stick.
90 var wPressed = keyboard.wKey.isPressed;
91 var aPressed = keyboard.aKey.isPressed;
92 var sPressed = keyboard.sKey.isPressed;
93 var dPressed = keyboard.dKey.isPressed;
94
95 if (aPressed)
96 {
97 //Debug.Log("A is pressed");
98 state.x -= 127;
99 }
100 if (dPressed)
101 {
102 //Debug.Log("D is pressed");
103 state.x += 127;
104 }
105 if (wPressed)
106 {
107 //Debug.Log("W is pressed");
108 state.y += 127;
109 }
110 if (sPressed)
111 {
112 //Debug.Log("S is pressed");
113 state.y -= 127;
114 }
115
116 // Map buttons to 1, 2, and 3.
117 if (keyboard.digit1Key.isPressed)
118 state.buttons |= 1 << 0;
119 if (keyboard.digit2Key.isPressed)
120 state.buttons |= 1 << 1;
121 if (keyboard.digit3Key.isPressed)
122 state.buttons |= 1 << 2;
123
124 InputSystem.QueueStateEvent(this, state);
